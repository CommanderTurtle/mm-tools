cmake_minimum_required(VERSION 3.20)
project(wemm_native LANGUAGES C CXX)
include(GNUInstallDirs)
find_package(Python3 3.10 COMPONENTS Interpreter REQUIRED)
find_package(Threads REQUIRED)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_BUILD_RPATH_USE_ORIGIN ON)
set(CMAKE_INSTALL_RPATH "$ORIGIN;$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH OFF)
set(LLAMA_CPP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/llama.cpp" CACHE PATH "Pinned upstream source")
execute_process(
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify.py"
            --llama-source "${LLAMA_CPP_DIR}"
    RESULT_VARIABLE verification_status)
if(NOT verification_status EQUAL 0)
    message(FATAL_ERROR "Native runtime or pinned llama.cpp verification failed")
endif()

set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_CPP_DIR}" llama)

foreach(target wemm-native wemm-native-audio)
    if(target STREQUAL "wemm-native")
        add_executable(${target} src/native_runner.cpp)
    else()
        add_executable(${target} src/native_audio_tool.cpp)
    endif()
    target_include_directories(${target} PRIVATE include "${LLAMA_CPP_DIR}/vendor")
    target_link_libraries(${target} PRIVATE llama mtmd ggml-cpu Threads::Threads)
    install(TARGETS ${target} RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
endforeach()
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
    target_link_libraries(ggml PUBLIC stdc++fs)
    target_link_options(ggml PUBLIC "-Wl,--exclude-libs,libstdc++fs.a")
endif()
install(FILES README.md THIRD_PARTY_NOTICES.md llama.cpp.lock.json source_manifest.json
        DESTINATION "${CMAKE_INSTALL_DATADIR}/wemm-native")
install(DIRECTORY include/third_party/ DESTINATION "${CMAKE_INSTALL_DATADIR}/wemm-native/third_party")
